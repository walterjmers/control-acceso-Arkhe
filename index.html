// ============================================================ // CONTROL DE ACCESO - LA PIEDRA BRUTA // ============================================================ function doGet(e) { try { // -------------------------------------------------------- // 1. COMPROBAR QUE LLEGÓ UN CÓDIGO // -------------------------------------------------------- if (!e || !e.parameter || !e.parameter.codigo) { return responderJSON({ estado: "error", mensaje: "No se recibió ningún código." }); } // Código recibido desde el lector QR var codigoRecibido = String(e.parameter.codigo).trim(); // -------------------------------------------------------- // 2. ABRIR LA PLANILLA // -------------------------------------------------------- var spreadsheet = SpreadsheetApp.getActiveSpreadsheet(); // Nombre exacto de tu hoja según la captura var sheet = spreadsheet.getSheetByName("Form_Responses"); if (!sheet) { return responderJSON({ estado: "error", mensaje: "No se encontró la hoja Form_Responses." }); } // -------------------------------------------------------- // 3. COLUMNAS DE TU PLANILLA // -------------------------------------------------------- var COL_NOMBRE = 2; // B var COL_DIA = 5; // E var COL_CANTIDAD = 6; // F var COL_PAGO = 7; // G var COL_CODIGO = 9; // I var COL_USADO = 10; // J // -------------------------------------------------------- // 4. LEER LOS DATOS // -------------------------------------------------------- var ultimaFila = sheet.getLastRow(); if (ultimaFila < 2) { return responderJSON({ estado: "no_encontrado" }); } var datos = sheet .getRange(2, 1, ultimaFila - 1, 10) .getValues(); // -------------------------------------------------------- // 5. BUSCAR EL CÓDIGO // -------------------------------------------------------- for (var i = 0; i < datos.length; i++) { var codigoGuardado = String(datos[i][COL_CODIGO - 1]).trim(); if (codigoGuardado === codigoRecibido) { var filaReal = i + 2; var nombre = datos[i][COL_NOMBRE - 1]; var dia = datos[i][COL_DIA - 1]; var cantidad = datos[i][COL_CANTIDAD - 1]; var pago = datos[i][COL_PAGO - 1]; var usado = datos[i][COL_USADO - 1]; // ---------------------------------------------------- // 6. COMPROBAR QUE ESTÉ PAGADO // ---------------------------------------------------- if ( String(pago).trim().toLowerCase() !== "sí" && String(pago).trim().toLowerCase() !== "si" ) { return responderJSON({ estado: "no_pagado", nombre: nombre, cantidad: cantidad, dia: dia }); } // ---------------------------------------------------- // 7. COMPROBAR SI YA FUE UTILIZADO // ---------------------------------------------------- if ( String(usado).trim().toLowerCase() === "sí" || String(usado).trim().toLowerCase() === "si" || String(usado).trim().toLowerCase() === "usado" ) { return responderJSON({ estado: "usado", nombre: nombre, cantidad: cantidad, dia: dia }); } // ---------------------------------------------------- // 8. MARCAR LA ENTRADA COMO UTILIZADA // ---------------------------------------------------- sheet .getRange(filaReal, COL_USADO) .setValue("Sí"); // ---------------------------------------------------- // 9. DEVOLVER ENTRADA VÁLIDA // ---------------------------------------------------- return responderJSON({ estado: "valido", nombre: nombre, cantidad: cantidad, dia: dia, codigo: codigoRecibido }); } } // -------------------------------------------------------- // 10. SI EL CÓDIGO NO EXISTE // -------------------------------------------------------- return responderJSON({ estado: "no_encontrado" }); } catch (error) { return responderJSON({ estado: "error", mensaje: error.toString() }); } } // ============================================================ // FUNCIÓN PARA RESPONDER AL LECTOR QR // ============================================================ function responderJSON(objeto) { return ContentService .createTextOutput(JSON.stringify(objeto)) .setMimeType(ContentService.MimeType.JSON); }
